() => {
  'use strict';

  /* ── DOM References ── */
  var P  = document.getElementById('__omnitasker_panel');
  var TB = document.getElementById('__ot_titlebar');
  var BD = document.getElementById('__ot_body');
  var CB = document.getElementById('__ot_collapse_btn');
  var ORB = document.getElementById('__ot_orb');
  var PILL = document.getElementById('__ot_phase_pill');
  var PROG_W = document.getElementById('__ot_progress_wrap');
  var PROG_F = document.getElementById('__ot_progress_fill');
  var PROG_L = document.getElementById('__ot_progress_label');
  var HITL   = document.getElementById('__ot_hitl');
  var HITL_R = document.getElementById('__ot_hitl_reason');
  var COG_TH = document.getElementById('__ot_cog_thinking');
  var COG_TH_T = document.getElementById('__ot_cog_thinking_text');
  var COG_GO = document.getElementById('__ot_cog_goal');
  var COG_GO_T = document.getElementById('__ot_cog_goal_text');
  var COG_ME = document.getElementById('__ot_cog_memory');
  var COG_ME_T = document.getElementById('__ot_cog_memory_text');
  var COG_AC = document.getElementById('__ot_cog_actions');
  var COG_AC_L = document.getElementById('__ot_cog_actions_list');
  var TASKS_S = document.getElementById('__ot_tasks_section');
  var TASKS_L = document.getElementById('__ot_tasks_list');
  var LOGS   = document.getElementById('__ot_logs_box');
  var PAUSE  = document.getElementById('__ot_pause_btn');
  var STOP   = document.getElementById('__ot_stop_btn');
  var SEND   = document.getElementById('__ot_send_btn');
  var S_INP  = document.getElementById('__ot_suggest_input');
  var FAIL   = document.getElementById('__ot_failed');
  var FAIL_P = document.getElementById('__ot_failed_pre');
  var COPY   = document.getElementById('__ot_copy_btn');

  if (!P) return;

  // Fix #9: Tie init flag to the panel DOM element, not window.
  // On SPAs (React/Vue), body gets wiped but window flag persists,
  // causing re-injected panels to have zero event listeners (zombie trap).
  if (P.__ot_initialized) {
    // Panel already has listeners attached — just update state if available
    try {
      var initEl = document.getElementById('__omnitasker_state');
      if (initEl && initEl.dataset.state && initEl.dataset.state !== '{}') {
        if (window.__ot_updateUI) {
          window.__ot_updateUI(JSON.parse(initEl.dataset.state));
        }
      }
    } catch(ex) {}
    return;
  }
  P.__ot_initialized = true;
  window.__omnitasker_js_initialized = true;

  /* ── Panel State & Transitions (Starts Expanded by Default, Collapsible to Rounded Square Pill) ── */
  var collapsed = false;

  function expandPanel() {
    collapsed = false;
    P.classList.remove('__ot_collapsed_ball');
    P.style.setProperty('width', '380px', 'important');
    P.style.setProperty('height', 'auto', 'important');
    P.style.setProperty('min-width', '380px', 'important');
    P.style.setProperty('max-width', '380px', 'important');
    P.style.setProperty('min-height', 'auto', 'important');
    P.style.setProperty('max-height', 'none', 'important');
    P.style.setProperty('border-radius', '16px', 'important');
    P.style.setProperty('cursor', 'default', 'important');
    P.style.setProperty('bottom', '20px', 'important');
    P.style.setProperty('right', '20px', 'important');
    P.style.setProperty('left', 'auto', 'important');
    P.style.setProperty('top', 'auto', 'important');
    if (BD) BD.style.display = '';
    if (CB) CB.title = 'Collapse panel';
  }

  function collapsePanel() {
    collapsed = true;
    P.classList.add('__ot_collapsed_ball');
    P.style.setProperty('width', '52px', 'important');
    P.style.setProperty('height', '52px', 'important');
    P.style.setProperty('min-width', '52px', 'important');
    P.style.setProperty('max-width', '52px', 'important');
    P.style.setProperty('min-height', '52px', 'important');
    P.style.setProperty('max-height', '52px', 'important');
    P.style.setProperty('border-radius', '14px', 'important');
    P.style.setProperty('cursor', 'pointer', 'important');
    P.style.setProperty('bottom', '20px', 'important');
    P.style.setProperty('right', '20px', 'important');
    P.style.setProperty('left', 'auto', 'important');
    P.style.setProperty('top', 'auto', 'important');
    if (BD) BD.style.display = 'none';
    if (CB) CB.title = 'Expand panel';

    // Apply active phase accents to the pill immediately upon collapse
    if (lastState && lastState.phase) {
      P.classList.remove('--phase-recon', '--phase-exec', '--phase-paused', '--phase-error', '--phase-done');
      if (lastState.phase === 'recon') P.classList.add('--phase-recon');
      else if (lastState.phase === 'executing') P.classList.add('--phase-exec');
      else if (lastState.phase === 'paused_hitl') P.classList.add('--phase-paused');
      else if (lastState.phase === 'error') P.classList.add('--phase-error');
      else if (lastState.phase === 'done') P.classList.add('--phase-done');
    }
  }

  // Restore saved position or default to bottom-right
  try {
    var rawPos = window.localStorage.getItem('__ot_panel_pos');
    if (rawPos) {
      var pos = JSON.parse(rawPos);
      if (typeof pos.left === 'number' && typeof pos.top === 'number') {
        var w = collapsed ? 52 : 380;
        var h = 52;
        var nx = Math.max(8, Math.min(pos.left, window.innerWidth - w - 8));
        var ny = Math.max(8, Math.min(pos.top, window.innerHeight - h - 8));
        P.style.setProperty('left', nx + 'px', 'important');
        P.style.setProperty('top', ny + 'px', 'important');
        P.style.setProperty('right', 'auto', 'important');
        P.style.setProperty('bottom', 'auto', 'important');
      }
    }
  } catch(ex) {}

  /* ── Drag & Click Logic ── */
  var dragging = false, hasMoved = false, startMouseX = 0, startMouseY = 0, dX = 0, dY = 0;

  function onMouseDown(e) {
    if (e.button !== 0) return;
    if (!collapsed) {
      if (!TB || (!TB.contains(e.target) && e.target !== TB)) return;
      if (CB && CB.contains(e.target)) return;
    }
    dragging = true;
    hasMoved = false;
    startMouseX = e.clientX;
    startMouseY = e.clientY;
    var r = P.getBoundingClientRect();
    dX = e.clientX - r.left;
    dY = e.clientY - r.top;
  }

  P.addEventListener('mousedown', onMouseDown);

  document.addEventListener('mousemove', function(e) {
    if (!dragging || !P) return;
    if (Math.hypot(e.clientX - startMouseX, e.clientY - startMouseY) > 5) {
      hasMoved = true;
    }
    if (!hasMoved) return;

    var curW = P.offsetWidth || 52;
    var curH = P.offsetHeight || 52;
    var nx = e.clientX - dX;
    var ny = e.clientY - dY;
    nx = Math.max(8, Math.min(nx, window.innerWidth - curW - 8));
    ny = Math.max(8, Math.min(ny, window.innerHeight - curH - 8));

    P.style.setProperty('left', nx + 'px', 'important');
    P.style.setProperty('top', ny + 'px', 'important');
    P.style.setProperty('right', 'auto', 'important');
    P.style.setProperty('bottom', 'auto', 'important');
    e.preventDefault();
  });

  document.addEventListener('mouseup', function() {
    if (dragging) {
      dragging = false;
      if (hasMoved) {
        try {
          var r = P.getBoundingClientRect();
          window.localStorage.setItem('__ot_panel_pos', JSON.stringify({ left: Math.round(r.left), top: Math.round(r.top) }));
        } catch(ex) {}
      }
    }
  });

  P.addEventListener('click', function(e) {
    if (hasMoved) {
      hasMoved = false;
      e.stopPropagation();
      e.preventDefault();
      return;
    }
    if (collapsed) {
      expandPanel();
      e.stopPropagation();
    }
  });

  if (CB) {
    CB.addEventListener('click', function(e) {
      if (hasMoved) return;
      e.stopPropagation();
      if (collapsed) {
        expandPanel();
      } else {
        collapsePanel();
      }
    });
  }

  /* ── Command Dispatch ── */
  function sendCmd(action, payload) {
    var el = document.getElementById('__omnitasker_commands');
    if (el) {
      el.dataset.action = action || '';
      el.dataset.payload = payload || '';
    }
  }

  if (PAUSE) {
    PAUSE.addEventListener('click', function(e) {
      e.stopPropagation();
      if (PAUSE.dataset.paused === 'true') {
        sendCmd('resume');
        PAUSE.innerHTML = '⏯️ Pause';
        PAUSE.className = '__ot_btn --pause';
        PAUSE.dataset.paused = 'false';
      } else {
        sendCmd('pause');
        PAUSE.innerHTML = '🔓 Resume';
        PAUSE.className = '__ot_btn --resume';
        PAUSE.dataset.paused = 'true';
      }
    });
  }
  if (STOP) {
    STOP.addEventListener('click', function(e) { e.stopPropagation(); sendCmd('stop'); });
  }
  if (SEND) {
    SEND.addEventListener('click', function() {
      if (!S_INP) return;
      var v = S_INP.value.trim();
      if (v) { sendCmd('suggestion', v); S_INP.value = ''; }
    });
  }
  if (S_INP) {
    S_INP.addEventListener('keydown', function(e) {
      if (e.key === 'Enter') { e.preventDefault(); if (SEND) SEND.click(); }
    });
  }
  if (COPY) {
    COPY.addEventListener('click', function() {
      var code = (FAIL_P && FAIL_P.textContent) || '';
      try {
        navigator.clipboard.writeText(code).then(function() {
          COPY.textContent = '\u2713 Copied';
          setTimeout(function() { if (COPY) COPY.textContent = 'Copy'; }, 2000);
        });
      } catch(ex) {
        /* Fallback: select text */
        if (FAIL_P) {
          var range = document.createRange();
          range.selectNodeContents(FAIL_P);
          var sel = window.getSelection();
          sel.removeAllRanges();
          sel.addRange(range);
        }
      }
    });
  }

  /* ── Helper: escape HTML ── */
  function esc(s) { var d = document.createElement('div'); d.textContent = s; return d.innerHTML; }

  /* ── Status Icon Map ── */
  var TASK_ICONS = {
    pending: '\u25FB\uFE0F', in_progress: '\uD83D\uDD04',
    success: '\u2705', failed: '\u274C', retrying: '\uD83D\uDD01'
  };

  /* ── Phase Map ── */
  var PHASE_CFG = {
    idle:        { label: 'Idle',      orbCls: '',        pillCls: '' },
    recon:       { label: 'Recon',     orbCls: '--recon', pillCls: '--recon' },
    executing:   { label: 'Executing', orbCls: '--exec',  pillCls: '--exec' },
    paused_hitl: { label: 'Paused',    orbCls: '--paused',pillCls: '--paused' },
    done:        { label: 'Done',      orbCls: '--done',  pillCls: '--done' },
    error:       { label: 'Error',     orbCls: '--error', pillCls: '--error' }
  };

  /* ── Main UI Update ── */
  var lastJson = '';
  var lastState = null;

  function updateUI(s) {
    if (!s) return;
    lastState = s;
    /* Phase */
    var pc = PHASE_CFG[s.phase] || PHASE_CFG.idle;
    if (ORB) ORB.className = '__ot_orb ' + pc.orbCls;
    if (PILL) {
      PILL.className = '__ot_phase_pill ' + pc.pillCls;
      PILL.textContent = pc.label;
    }

    /* Synchronize phase classes on panel element so collapsed pill dynamically glows with the current phase */
    P.classList.remove('--phase-recon', '--phase-exec', '--phase-paused', '--phase-error', '--phase-done');
    if (s.phase === 'recon') P.classList.add('--phase-recon');
    else if (s.phase === 'executing') P.classList.add('--phase-exec');
    else if (s.phase === 'paused_hitl') P.classList.add('--phase-paused');
    else if (s.phase === 'error') P.classList.add('--phase-error');
    else if (s.phase === 'done') P.classList.add('--phase-done');

    /* Auto-expand on HITL or Error */
    if (collapsed && (s.phase === 'paused_hitl' || s.hitlReason || s.phase === 'error')) {
      expandPanel();
    }

    /* Progress */
    var showProg = (s.phase === 'executing' || s.phase === 'done') && s.totalTasks > 0;
    if (PROG_W) PROG_W.className = '__ot_progress_wrap' + (showProg ? ' --visible' : '');
    if (showProg && PROG_F && PROG_L) {
      var pct = s.phase === 'done' ? 100 : Math.round((s.progress || 0) * 100);
      PROG_F.style.width = pct + '%';
      if (s.phase === 'done') {
        PROG_F.classList.add('--done');
        PROG_L.textContent = '100% \u2014 All Tasks Completed';
      } else {
        PROG_F.classList.remove('--done');
        PROG_L.textContent = pct + '% \u2014 Task ' + ((s.currentTask || 0) + 1) + '/' + s.totalTasks;
      }
    }

    /* HITL Alert */
    if (HITL && HITL_R) {
      if (s.hitlReason) {
        HITL.className = '__ot_hitl --visible';
        HITL_R.textContent = s.hitlReason;
      } else {
        HITL.className = '__ot_hitl';
      }
    }

    /* Cognitive Cards */
    function showCog(el, textEl, val, isMono) {
      if (!el || !textEl) return;
      if (val) {
        el.className = '__ot_cog --visible';
        if (isMono) textEl.textContent = val;
        else textEl.textContent = val;
      } else {
        el.className = '__ot_cog';
      }
    }
    showCog(COG_TH, COG_TH_T, s.thinking, false);
    showCog(COG_GO, COG_GO_T, s.goal, false);
    showCog(COG_ME, COG_ME_T, s.memory, true);

    if (COG_AC && COG_AC_L) {
      if (s.actions && s.actions.length > 0) {
        COG_AC.className = '__ot_cog --visible';
        COG_AC_L.innerHTML = s.actions.map(function(a) { return '' + esc(a) + ''; }).join('');
      } else {
        COG_AC.className = '__ot_cog';
      }
    }

    /* Tasks */
    if (TASKS_S && TASKS_L) {
      if (s.tasks && s.tasks.length > 0) {
        TASKS_S.className = '__ot_tasks_section --visible';
        TASKS_L.innerHTML = s.tasks.map(function(t, i) {
          var icon = TASK_ICONS[t.status] || '\u2753';
          var cur = (i === (s.currentTask || 0) && s.phase === 'executing') ? ' --current' : '';
          return '
' + icon + '#' + t.id + ' ' + esc(t.desc) + '
';
        }).join('');
      } else {
        TASKS_S.className = '__ot_tasks_section';
      }
    }

    /* Logs */
    if (LOGS && s.logs && s.logs.length > 0) {
      var html = '';
      var slice = s.logs.slice(-30);
      for (var i = 0; i < slice.length; i++) {
        var lg = slice[i];
        var lcls = lg.l === 'success' ? ' --success' : lg.l === 'warn' ? ' --warn' : lg.l === 'error' ? ' --error' : '';
        html += '[' + esc(lg.t) + ']' + esc(lg.m) + '';
      }
      LOGS.innerHTML = html;
      LOGS.scrollTop = LOGS.scrollHeight;
    }

    /* Controls */
    if (PAUSE) {
      var isPaused = s.phase === 'paused_hitl';
      if (isPaused) {
        PAUSE.innerHTML = '🔓 Resume';
        PAUSE.className = '__ot_btn --resume';
        PAUSE.dataset.paused = 'true';
      } else {
        PAUSE.innerHTML = '⏯️ Pause';
        PAUSE.className = '__ot_btn --pause';
        PAUSE.dataset.paused = 'false';
      }
    }

    /* Failed Code */
    if (FAIL && FAIL_P) {
      if (s.failedCode) {
        FAIL.className = '__ot_failed --visible';
        FAIL_P.textContent = s.failedCode;
      } else {
        FAIL.className = '__ot_failed';
      }
    }
  }

  window.__ot_updateUI = updateUI;

  /* ── Polling Loop ── */
  setInterval(function() {
    try {
      var el = document.getElementById('__omnitasker_state');
      if (!el) return;
      var j = el.dataset.state;
      if (!j || j === '{}' || j === lastJson) return;
      lastJson = j;
      var st = JSON.parse(j);
      updateUI(st);
    } catch(e) { /* silently ignore parse errors */ }
  }, 750);

  /* ── Prevent panel clicks from propagating to page ── */
  P.addEventListener('mousedown', function(e) { e.stopPropagation(); });
  P.addEventListener('click',     function(e) { e.stopPropagation(); });
  P.addEventListener('keydown',   function(e) { e.stopPropagation(); });
  P.addEventListener('keyup',     function(e) { e.stopPropagation(); });

}
